         date = datetime.now(UTC).date().isoformat()

         base = f"{date}-{slugify(proposal.title)}"
         path = _unused_path(config.root_path, directory, base)

             "type: source",
             f"source_type: {proposal.source_type}",

             f"title: {json.dumps(proposal.title)}",
             f"retrieved: {date}",
             "---",
             "",
         ]
